ai_summary.py — full module to build today
import anthropic
from dotenv import load_dotenv
import os

load_dotenv()

def _build_prompt(metrics: dict, date_range: tuple) -> str:
    """Inject real metric values into the user prompt."""
    start, end = date_range
    return f"""
<date_range>{start} to {end}</date_range>

<metrics>
- Overall denial rate: {metrics['denial_rate']:.1f}%
- Days in A/R (average): {metrics['days_in_ar']:.0f} days
- Total $ at risk (denied claims): ${metrics['dollars_at_risk']:,.0f}
- Clean claim rate: {metrics['clean_claim_rate']:.1f}%
- First-pass rate: {metrics.get('first_pass_rate', 'N/A')}
- Top denial reason: {metrics.get('top_denial_reason', 'N/A')}
- Highest-denial payer: {metrics.get('top_denial_payer', 'N/A')}
</metrics>

<task>
Write a 4-6 sentence daily briefing for the billing manager.
Start with the single most important finding.
Mention any metric that exceeds a concerning threshold.
End with one specific, actionable recommendation.
Do not invent any numbers — use only the values above.
</task>
"""

def generate_briefing(metrics: dict, date_range: tuple) -> str:
    """Call Claude API and return the generated briefing text."""
    # Load system prompt from file (written in Week 2)
    prompt_path = "src/prompts/briefing_system_prompt.txt"
    with open(prompt_path) as f:
        system_prompt = f.read()

    client = anthropic.Anthropic(
        api_key=os.getenv("ANTHROPIC_API_KEY")
    )
    message = client.messages.create(
        model="claude-haiku-4-5-20251001",  # fast + cheap for daily summaries
        max_tokens=400,
        system=system_prompt,
        messages=[{
            "role": "user",
            "content": _build_prompt(metrics, date_range)
        }]
    )
    return message.content[0].text

def generate_briefing_safe(metrics: dict, date_range: tuple) -> str:
    """Wrapper with error handling — returns fallback message on API failure."""
    try:
        return generate_briefing(metrics, date_range)
    except Exception as e:
        return f"AI briefing unavailable ({type(e).__name__}). Review metrics manually."
🎯 Why claude-haiku for daily summaries?

Haiku is Anthropic's fastest and most cost-efficient model. For a 4-6 sentence briefing from structured data, it performs identically to Sonnet at a fraction of the cost. Use Sonnet for complex reasoning tasks. Use Haiku for fast, repetitive, structured outputs. This is a real cost-optimization decision that matters at scale.

def stream_briefing(metrics: dict, date_range: tuple):
    """Generator that yields text chunks for st.write_stream."""
    with open("src/prompts/briefing_system_prompt.txt") as f:
        system = f.read()
    client = anthropic.Anthropic(api_key=os.getenv("ANTHROPIC_API_KEY"))
    with client.messages.stream(
        model="claude-haiku-4-5-20251001",
        max_tokens=400,
        system=system,
        messages=[{"role":"user","content":_build_prompt(metrics,date_range)}]
    ) as stream:
        for text in stream.text_stream:
            yield text
