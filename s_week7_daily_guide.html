Storytelling with Data — Gestalt Principles
Six psychological principles that explain how humans perceive visual groups: proximity, similarity, continuity, closure, figure/ground, and common fate. Each one has a direct application in dashboard layout design.
Free · storytellingwithdata.com
🎨
NNG — F-Pattern Eye Tracking Research
Eye-tracking studies showing that users scan dashboards in an F-pattern — heavy attention top-left, decreasing as they move right and down. This means KPI tiles must go top-left, and the most critical chart belongs full-width below them.
Free · nngroup.com

src/charts.py — expanded apply_theme() using Plotly template
import plotly.graph_objects as go
import plotly.io as pio

# Design system constants
DS = {
    'navy':      '#1b2a4a',
    'cream':     '#f7f3ee',
    'paper':     '#faf8f5',
    'border':    '#e2ddd6',
    'muted':     '#7a7060',
    'green':     '#22c55e',
    'amber':     '#f59e0b',
    'orange':    '#f97316',
    'red':       '#ef4444',
    'font_serif':'Lora, Georgia, serif',
    'font_sans': 'IBM Plex Sans, Arial, sans-serif',
    'font_mono': 'IBM Plex Mono, monospace',
}

def apply_theme(fig, title: str = None, height: int = 320):
    """Apply ClaimPulse design system to any Plotly figure."""
    fig.update_layout(
        paper_bgcolor=DS['cream'],
        plot_bgcolor=DS['cream'],
        font=dict(family=DS['font_sans'], size=12, color=DS['navy']),
        title=dict(
            text=title, font=dict(family=DS['font_serif'],
            size=15, color=DS['navy']),
            x=0, xanchor='left', pad=dict(l=0, b=8)
        ),
        margin=dict(l=16, r=16, t=48, b=16),
        height=height,
        legend=dict(bgcolor=DS['cream'], bordercolor=DS['border'],
                    borderwidth=1, font=dict(size=11)),
        hovermode='x unified',
        hoverlabel=dict(bgcolor=DS['navy'], font_color='white',
                        font_size=12, bordercolor=DS['navy']),
    )
    fig.update_xaxes(
        gridcolor=DS['border'], linecolor=DS['border'],
        tickfont=dict(size=11, color=DS['muted']),
        zeroline=False
    )
    fig.update_yaxes(
        gridcolor=DS['border'], linecolor=DS['border'],
        tickfont=dict(size=11, color=DS['muted']),
        zeroline=False
    )
    return fig
