def get_prior_period(df, date_range):
    """Return a DataFrame for the period immediately before date_range."""
    start, end = date_range
    duration = (end - start).days
    prior_end = start - pd.Timedelta(days=1)
    prior_start = prior_end - pd.Timedelta(days=duration)
    return df[
        (df['service_date'].dt.date >= prior_start) &
        (df['service_date'].dt.date <= prior_end)
    ]

# In app.py — compute deltas
prior_df = get_prior_period(df, date_range)

# Build delta strings
def delta_str(current, prior, pct=True, invert=False):
    if prior is None or prior == 0: return None
    diff = current - prior
    sym = '▲' if diff > 0 else '▼'
    val = f"{abs(diff):.1f}{'%' if pct else ''}"
    return f"{sym} {val} vs prior period"

k1.metric("Denial Rate", f"{metrics['denial_rate']:.1f}%",
          delta=delta_str(metrics['denial_rate'],
                          prior_metrics['denial_rate'] if prior_metrics else None),
          delta_color="inverse")

def payer_trend_chart(df: pd.DataFrame, top_n: int = 4):
    """Multi-line denial trend — one line per top payer."""
    # Get top N payers by volume
    top_payers = (df.groupby('payer_name')['claim_id']
                  .count().nlargest(top_n).index.tolist())
    df_top = df[df['payer_name'].isin(top_payers)].copy()
    df_top['week'] = df_top['service_date'].dt.to_period('W').astype(str)

    weekly = df_top.groupby(['week', 'payer_name']).agg(
        total=('claim_id', 'count'),
        denied=('status', lambda x: (x=='denied').sum())
    ).reset_index()
    weekly['denial_rate'] = weekly['denied']/weekly['total']*100

    fig = px.line(weekly, x='week', y='denial_rate',
                  color='payer_name', markers=True,
                  title='Denial Rate Trend by Payer')
    return apply_theme(fig)

from openpyxl.styles import Font, PatternFill, Alignment
from openpyxl.utils import get_column_letter

def to_excel_report(filtered_df, metrics, payer_stats, top_cpts) -> bytes:
    output = io.BytesIO()
    with pd.ExcelWriter(output, engine='openpyxl') as writer:
        # Sheet 1: Summary metrics
        summary = pd.DataFrame([{
            'Metric': k, 'Value': v
        } for k, v in metrics.items()])
        summary.to_excel(writer, sheet_name='Summary', index=False)

        # Sheet 2: Filtered claim data
        filtered_df.to_excel(writer, sheet_name='Claim Data', index=False)

        # Sheet 3: Payer breakdown
        payer_stats.to_excel(writer, sheet_name='Payer Breakdown', index=False)

        # Sheet 4: Top denied CPTs
        top_cpts.to_excel(writer, sheet_name='Top Denied CPTs', index=False)

        # Style headers across all sheets
        wb = writer.book
        header_fill = PatternFill('solid', fgColor='1b2a4a')
        header_font = Font(color='FFFFFF', bold=True)
        for ws in wb.worksheets:
            for cell in ws[1]:
                cell.fill = header_fill
                cell.font = header_font

    return output.getvalue()
